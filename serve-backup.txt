
type ChatRequestBody = {
  message?: unknown;
  model?: unknown;
};

type ClaudeResponse = {
  error?: {
    message?: string;
  };
  content?: Array<{
    type?: string;
    text?: string;
  }>;
};

type GeminiResponse = {
  error?: {
    message?: string;
  };
  candidates?: Array<{
    content?: {
      parts?: Array<{
        text?: string;
      }>;
    };
  }>;
};

const PORT = Number(Bun.env.PORT || 3000);
const html = Bun.file("public/index.html");

Bun.serve({
  port: PORT,

  async fetch(req) {
    const url = new URL(req.url);

    // Health Check
    if (url.pathname === "/api/health") {
      return Response.json({
        status: "online",
        app: "USMANI AI",
        providers: ["Anthropic Claude", "Google Gemini"]
      });
    }

    // Chat API
    if (url.pathname === "/api/chat" && req.method === "POST") {
      try {
        const body = (await req.json()) as ChatRequestBody;

        const message =
          typeof body.message === "string"
            ? body.message.trim()
            : "";

        const model =
          typeof body.model === "string" && body.model.trim()
            ? body.model.trim()
            : "claude-sonnet-4-5";

        if (!message) {
          return Response.json(
            { error: "Message empty hai." },
            { status: 400 }
          );
        }

        // GOOGLE GEMINI
        if (model === "gemini-3.8-flash") {
          const apiKey = Bun.env.GEMINI_API_KEY;

          if (!apiKey) {
            return Response.json(
              { error: "GEMINI_API_KEY .env mein nahi mili." },
              { status: 500 }
            );
          }

          const response = await fetch(
            `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent`,
            {
              method: "POST",
              headers: {
                "x-goog-api-key": apiKey,
                "Content-Type": "application/json"
              },
              body: JSON.stringify({
                contents: [
                  {
                    role: "user",
                    parts: [{ text: message }]
                  }
                ]
              })
            }
          );

          const data = (await response.json()) as GeminiResponse;

          if (!response.ok) {
            return Response.json(
              {
                error: data.error?.message || "Gemini API error"
              },
              { status: response.status }
            );
          }

          const answer = (data.candidates || [])
            .flatMap((candidate) => candidate.content?.parts || [])
            .map((part) => part.text || "")
            .join("\n");

          return Response.json({
            answer,
            provider: "Google Gemini",
            model
          });
        }

        // ANTHROPIC CLAUDE
        const allowedClaudeModels = [
          "claude-sonnet-4-5"
        ];

        if (!allowedClaudeModels.includes(model)) {
          return Response.json(
            { error: "Selected model abhi supported nahi hai." },
            { status: 400 }
          );
        }

        const apiKey = Bun.env.ANTHROPIC_API_KEY;

        if (!apiKey) {
          return Response.json(
            { error: "ANTHROPIC_API_KEY .env mein nahi mili." },
            { status: 500 }
          );
        }

        const response = await fetch(
          "https://api.anthropic.com/v1/messages",
          {
            method: "POST",
            headers: {
              "x-api-key": apiKey,
              "anthropic-version": "2023-06-01",
              "content-type": "application/json"
            },
            body: JSON.stringify({
              model,
              max_tokens: 2048,
              messages: [
                {
                  role: "user",
                  content: message
                }
              ]
            })
          }
        );

        const data = (await response.json()) as ClaudeResponse;

        if (!response.ok) {
          return Response.json(
            {
              error: data.error?.message || "Claude API error"
            },
            { status: response.status }
          );
        }

        const answer = (data.content || [])
          .filter((item) => item.type === "text")
          .map((item) => item.text || "")
          .join("\n");

        return Response.json({
          answer,
          provider: "Anthropic Claude",
          model
        });

      } catch (error) {
        console.error("Chat API Error:", error);

        return Response.json(
          { error: "Server request failed. Terminal check karein." },
          { status: 500 }
        );
      }
    }

    // Homepage
    if (url.pathname === "/" || url.pathname === "/index.html") {
      return new Response(html, {
        headers: {
          "Content-Type": "text/html; charset=utf-8"
        }
      });
    }

    // Unknown API Routes
    if (url.pathname.startsWith("/api/")) {
      return Response.json(
        { error: "API route not found" },
        { status: 404 }
      );
    }

    return new Response("Not Found", { status: 404 });
  }
});

console.log(`USMANI AI running at http://localhost:${PORT}`);
